PIN = "ca41c79426a90a57e78ca9554a993db326461737"
root = pathlib.Path(tempfile.mkdtemp(prefix="nba_ws5_"))
req = urllib.request.Request("https://codeload.github.com/royalexx/pickem-auditor/zip/"+PIN, headers={"User-Agent":"NBA-WS5-Pinned-Engineering"})
with urllib.request.urlopen(req, timeout=90) as response:
    archive = zipfile.ZipFile(io.BytesIO(response.read()))
for item in archive.infolist():
    parts = pathlib.PurePosixPath(item.filename).parts
    if len(parts)<2 or ".." in parts: continue
    path = root.joinpath(*parts[1:])
    if item.is_dir(): path.mkdir(parents=True, exist_ok=True)
    else:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(archive.read(item))
